\section{Integrali multipli}
    \subsection{Definizione di area e insieme misurabile su \texorpdfstring{$\mathbb{R}^{n}$}{R alla n}}
        Sia $I$ un qualsiasi intervallo aperto, chiuso o semichiuso e limitato in $\mathbb{R}$ e sia $\mathcal{L}(I) = b - a$ la ``misura'' di tale intervallo.
        Sia allora il volume di un generico cuboide rettangolare $C \subseteq \mathbb{R}^{n}, \; C = I_{1} \times ... \times I_{n}$, definito come
        \[
            \text{vol}(C) = \prod_{i = 1}^{n} \mathcal{L}(I_{i})
        \]
        Allora, dato un qualsiasi sottoinsieme limitato $D \subseteq \mathbb{R}^{n}$ si definisce l'ipervolume esterno (o misura di Peano-Jordan esterna) di $D$ come
        \[
            \lambda^{*}(D) = \inf{\left\{ \sum_{k = 1}^{p} \text{vol}(C_{k}) \st \{C_{k}\}_{k \leq p} \; \textnormal{successione finita di cuboidi tali che} \; D \subseteq \bigcup_{k = 1}^{p} C_{k} \right\}}
        \]
        Mentre l'ipervolume interno (o misura di Peano-Jordan interna) di $D$ è dato da
        \[
            \lambda_{*}(D) = \sup \left\{ \sum_{k = 1}^{p} \text{vol}(C_k) \st \{C_k\}_{k \leq p} \; \textnormal{successione finita di cuboidi disgiunti tali che} \; \bigcup_{k = 1}^{p} C_k \subseteq D \right\}
        \]


    \subsection{Integrale di Riemann in \texorpdfstring{$\mathbb{R}^{2}$}{R2}}
        \subsubsection{Definizione e proprietà dell'integrale}
        Sia $D \subseteq \mathbb{R}^{2}$. Si dicono, rispettivamente, area interna ed area esterna di $D$
        \[
            A^{-}(D) = \lambda_{*}(D) \quad \: A^{+}(D) = \lambda^{*}(D)
        \]
        $D \subseteq \mathbb{R}^{2}$ si dice misurabile qualora
        \[
            A^{-}(D) = A^{+}(D) = A(D) = |D|
        \]
        Inoltre, se e solo se $|\partial D| = 0$ si ha che $D$ è misurabile (criterio di Jordan-Lebesgue). 
        Dato un insieme $D \subseteq \mathbb{R}^{2}$ misurabile e una funzione $f: D \rightarrow \mathbb{R}$, posta una partizione $P(D, \delta) = (R_{i})_{i \in \mathbb{N}}$ tale che $R_{i} = I_{i, 1} \times I_{i, 2}$, $\displaystyle D \subseteq \bigcup_{i} R_{i}$ ed infine $\forall R_{i}, \; \max\{\Delta x_{i}, \Delta y_{i}\} \leq \delta \; \land \; R_{i} \cap D \neq \emptyset$,
        si dice somma di Riemann di $f$ lungo la partizione $P(D, \delta)$ la seguente espressione
        \[
            S(f, P(D, \delta)) = \sum_{i} f(x_i, y_i) \Delta x_i \Delta y_i
        \]
        dove $(x_i, y_i) \in R_{i}$ è un generico punto del rettangolo, mentre $\Delta x_i$ e $\Delta y_i$ non sono altro che le dimensioni del rettangolo: $\Delta x_i = \mathcal{L}(I_{i, 1})$, $\Delta y_i = \mathcal{L}(I_{i, 2})$.

        Se invece si prende il limite della somma di Riemann all'annullarsi della granularità della partizione $P(D, \delta)$ si ottiene l'integrale di Riemann
        \[
            \lim_{\delta \rightarrow 0} S(f, P(D, \delta)) = \lim_{\delta \rightarrow 0} \sum_{i}f(x_i, y_i) \Delta x_i \Delta y_i = \int_{D} f(x, y) dx dy
        \]

        \begin{enumerate}[label=(\arabic*)]
            \item Linearità dell'integrale: $\displaystyle \int_{D} (\alpha f + \beta g) = \alpha \int_{D} f + \beta \int_{D} g$
            \item Disuguaglianza dell'integrale: $\displaystyle \left| \int_{D} f \right| \leq \int_{D} |f| $
            \item Monotonia dell'integrale: $\displaystyle f \leq g \; \textnormal{in} \; D \implies \int_{D} f \leq \int_{D} g$
            \item Additività rispetto al dominio: $\displaystyle \mathring{D_1} \cap \mathring{D_2} = \emptyset \implies \int_{D_1 \cup D_2} f = \int_{D_1} f + \int_{D_2} f$
        \end{enumerate}


        \subsubsection{Formule di riduzione}
        \paragraph*{Formula di riduzione sui rettangoli.}
            Sia $D \subseteq \mathbb{R}^{2}$ nella forma $D = I_{1} \times I_{2}$ con $\partial I_{1} = \{a, b\}$ e $\partial I_{2} = \{c, d\}$, $-\infty < a < b < +\infty$, $-\infty < c < d < +\infty$.
            Allora
            \[
                \int_{D} f = \int_{a}^{b} \left( \int_{c}^{d} f(x, y) dy \right) dx = \int_{c}^{d} \left( \int_{a}^{b} f(x, y) dx \right) dy
            \]

        \paragraph*{Formula di riduzione su domini semplici rispetto all'asse $y$.}
            Sia $D \subseteq \mathbb{R}^{2}$ nella forma $D = \{(x, y) \in \mathbb{R}^{2} \st a \leq x \leq b, \; g_{1}(x) \leq y \leq g_{2}(x)\}$ dove $-\infty < a < b < +\infty$ e $g_{1}, g_{2}$ sono funzioni reali a variabile reale continue.
            Allora
            \[
                \int_{D} f = \int_{a}^{b} \left( \int_{g_{1}(x)}^{g_{2}(x)} f(x, y) dy \right) dx
            \]

        \paragraph*{Formula di riduzione su domini semplici rispetto all'asse $x$.}
            Sia $D \subseteq \mathbb{R}^{2}$ nella forma $D = \{(x, y) \in \mathbb{R}^{2} \st a \leq y \leq b, \; h_{1}(y) \leq x \leq h_{2}(y)\}$ dove $-\infty < a < b < +\infty$ e $h_{1}, h_{2}$ sono funzioni reali a variabile reale continue.
            Allora
            \[
                \int_{D} f = \int_{a}^{b} \left( \int_{h_{1}(y)}^{h_{2}(y)} f(x, y) dx \right) dy
            \]

        \subsubsection{Cambi di variabile}
            Data una funzione $f: D \subseteq \mathbb{R}^{2} \rightarrow \mathbb{R}$, sia $\Phi: D' \subseteq \mathbb{R}^{2} \rightarrow D$ una biezione di classe $C^{1}$, definita su un dominio $D'$ misurabile, tale per cui $\forall (x, y) \in \mathring{D'}, \; \det(J_{\Phi}) \neq 0$.
            Allora
            \[
                \int_{D} f = \int_{D'} f(\Phi) \; |\det(J_{\Phi})|
            \]

    \subsection{Integrale di Riemann in \texorpdfstring{$\mathbb{R}^{3}$}{R3}}
        \subsubsection{Definizione e proprietà dell'integrale}
        Sia $D \subseteq \mathbb{R}^{3}$. Si dicono, rispettivamente, volume interno e volume esterno di $D$
        \[
            V^{-}(D) = \lambda(D) \quad V^{+}(D) = \lambda^{*}(D)
        \]
        $D \subseteq \mathbb{R}^{3}$ si dice misurabile qualora
        \[
            V^{-}(D) = V^{+}(D) = V(D) = |D|
        \]

        Dato un insieme $D \subseteq \mathbb{R}^{3}$ misurabile e una funzione continua $f: D \rightarrow \mathbb{R}$, posta una partizione $P(D, \delta) = (C_{i})_{i \in \mathbb{N}}$ tale per cui $C_{i} = I_{i, 1} \times I_{i, 2} \times I_{i, 3}$, $D \subseteq \displaystyle \bigcup_{i} C_{i}$ ed infine $\forall C_{i}, \; \max\{\Delta x_{i}, \Delta y_{i}, \Delta z_{i}\} < \delta \; \land \; C_{i} \cap D \neq \emptyset$,
        si dice somma di Riemann di $f$ lungo la partizione $P(D, \delta)$ la seguente espressione
        \[
            S(f, P(D, \delta)) = \sum_{i} f(x_i, y_i, z_i) \Delta x_i \Delta y_i \Delta z_i
        \]
        dove $(x_i, y_i, z_i) \in C_{i}$ è un generico punto del cuboide, mentre $\Delta x_i$, $\Delta y_i$ e $\Delta z_i$ non sono altro che le dimensioni del cuboide: $\Delta x_i = \mathcal{L}(I_{i, 1})$, $\Delta x_i = \mathcal{L}(I_{i, 1})$, $\Delta x_i = \mathcal{L}(I_{i, 1})$

        Se invece si prende il limite della somma di Riemann all'annullarsi della granularità della partizione $P(D, \delta)$ si ottiene l'integrale di Riemann
        \[
            \lim_{\delta \rightarrow 0} S(f, P(D, \delta)) = \lim_{\delta \rightarrow 0} \sum_{i}f(x_i, y_i, z_i) \Delta x_i \Delta y_i \Delta z_i = \int_{D} f(x, y, z) dx dy dz
        \]

        \begin{enumerate}[label=(\arabic*)]
            \item Linearità dell'integrale: $\displaystyle \int_{D} (\alpha f + \beta g) = \alpha \int_{D} f + \beta \int_{D} g$
            \item Disuguaglianza dell'integrale: $\displaystyle \left| \int_{D} f \right| \leq \int_{D} |f| $
            \item Monotonia dell'integrale: $\displaystyle f \leq g \; \textnormal{in} \; D \implies \int_{D} f \leq \int_{D} g$
            \item Additività rispetto al dominio: $\displaystyle \mathring{D_1} \cap \mathring{D_2} = \emptyset \implies \int_{D_1 \cup D_2} f = \int_{D_1} f + \int_{D_2} f$
        \end{enumerate}


        \subsubsection{Formule di riduzione}
            \paragraph*{Formula di riduzione su domini normali rispetto all'asse $z$.}
                Sia $D \subseteq \mathbb{R}^{3}$ nella forma $D = \{(x, y, z) \in \mathbb{R}^{3} \st (x, y) \in E \subseteq \mathbb{R}^{2}, \; g_{1}(x, y) \leq z \leq g_{2}(x, y)\}$ dove $E$ è misurabile e $g_{1}, g_{2}$ sono funzioni continue in $E$.
                Allora
                \[
                    \int_{D} f = \int_{E} \left( \int_{g_{1}(x, y)}^{g_{2}(x, y)} f(x, y, z) dz \right)
                \]

            \paragraph*{Formula di riduzione su domini normali rispetto all'asse $y$.}
                Sia $D \subseteq \mathbb{R}^{3}$ nella forma $D = \{(x, y, z) \in \mathbb{R}^{3} \st (x, z) \in E \subseteq \mathbb{R}^{2}, \; g_{1}(x, z) \leq y \leq g_{2}(x, z)\}$ dove $E$ è misurabile e $g_{1}, g_{2}$ sono funzioni continue in $E$.
                Allora
                \[
                    \int_{D} f = \int_{E} \left( \int_{g_{1}(x, z)}^{g_{2}(x, z)} f(x, y, z) dy \right)
                \]

            \paragraph*{Formula di riduzione su domini normali rispetto all'asse $x$.}
                Sia $D \subseteq \mathbb{R}^{3}$ nella forma $D = \{(x, y, z) \in \mathbb{R}^{3} \st (y, z) \in E \subseteq \mathbb{R}^{2}, \; g_{1}(y, z) \leq x \leq g_{2}(y, z)\}$ dove $E$ è misurabile e $g_{1}, g_{2}$ sono funzioni continue in $E$.
                Allora
                \[
                    \int_{D} f = \int_{E} \left( \int_{g_{1}(y, z)}^{g_{2}(y, z)} f(x, y, z) dx \right)
                \]

            \paragraph*{Formula di riduzione su domini stratificati rispetto all'asse $z$.}
                Sia $D \subseteq \mathbb{R}^{3}$ nella forma $D = \{(x, y, z) \in \mathbb{R}^{3} \st a \leq z \leq b, \; (x, y) \in E_{z} \subseteq \mathbb{R}^{2}\}$ dove $-\infty < a < b < +\infty$ e $E_{z} = \{ (x, y) \in \mathbb{R}^{2} \st (x, y, z) \in D\}$.
                Allora
                \[
                    \int_{D} f = \int_{a}^{b} \left( \int_{E_{z}} f(x, y, z) \right) dz
                \]

            \paragraph*{Formula di riduzione su domini stratificati rispetto all'asse $y$.}
                Sia $D \subseteq \mathbb{R}^{3}$ nella forma $D = \{(x, y, z) \in \mathbb{R}^{3} \st a \leq y \leq b, \; (x, z) \in E_{y} \subseteq \mathbb{R}^{2}\}$ dove $-\infty < a < b < +\infty$ e $E_{y} = \{ (x, z) \in \mathbb{R}^{2} \st (x, y, z) \in D\}$.
                Allora
                \[
                    \int_{D} f = \int_{a}^{b} \left( \int_{E_{y}} f(x, y, z) \right) dy
                \]

            \paragraph*{Formula di riduzione su domini stratificati rispetto all'asse $x$.}
                Sia $D \subseteq \mathbb{R}^{3}$ nella forma $D = \{(x, y, z) \in \mathbb{R}^{3} \st a \leq x \leq b, \; (y, z) \in E_{x} \subseteq \mathbb{R}^{2}\}$ dove $-\infty < a < b < +\infty$ e $E_{x} = \{ (y, z) \in \mathbb{R}^{2} \st (x, y, z) \in D\}$.
                Allora
                \[
                    \int_{D} f = \int_{a}^{b} \left( \int_{E_{x}} f(x, y, z) \right) dx
                \]

        \subsubsection{Cambi di variabile}
            Data una funzione $f: D \subseteq \mathbb{R}^{3} \rightarrow \mathbb{R}$, sia $\Phi: D' \subseteq \mathbb{R}^{3} \rightarrow D$ una biezione di classe $C^{1}$, definita su un dominio $D'$ misurabile, tale per cui $\forall (x, y, z) \in \mathring{D'}, \; \det(J_{\Phi}) \neq 0$.
            Allora
            \[
                \int_{D} f = \int_{D'} f(\Phi) |\det(J_{\Phi})|
            \]

            Una varietà di cambi di variabile particolarmente utilizzati sono le cooridinate cilindriche e sferiche.
            Le coordinate cilindriche (rispetto a uno dei tre assi) sono coordinate nella forma
            \[
                \Phi: [0, +\infty) \times \mathbb{R} \times \mathbb{R} \rightarrow \mathbb{R}^{3}
            \]
            \[
                (r, \theta, t) \mapsto (r \cos(\theta), r \sin (\theta), t), \; \text{rispetto all'asse z}
            \]
            \[
                (r, \theta, t) \mapsto (r \cos(\theta), t, r \sin (\theta)), \; \text{rispetto all'asse y}
            \]
            \[
                (r, \theta, t) \mapsto (t, r \cos(\theta), r \sin(\theta)), \; \text{rispetto all'asse x}
            \]
            \[
                \det(J_{\Phi}) = r
            \]
            Le coordinate sferiche invece sono coordinate nella forma
            \[
                \Phi: [0, +\infty) \times \mathbb{R} \times \mathbb{R} \rightarrow \mathbb{R}^{3}
            \]
            \[
                (r, \theta, \varphi) \mapsto (r \sin(\varphi)\cos(\theta), r \sin(\varphi)\sin(\theta), r \cos(\varphi))
            \]
            \[
                \det(J_{\Phi}) = r^{2} \sin(\varphi)
            \]

        \subsubsection{Solidi di rotazione}
            Data una superficie $S \subseteq \mathbb{R}^{2}$ giacente sul piano $yz$ è possibile calcolare il volume del solido ottenuto ruotando la superficie attorno all'asse $z$ nel seguente modo:
            si esegua il cambio di variabile
            \[
                \Phi: [0, 2 \pi) \times S \rightarrow \mathbb{R}^{3}
            \]
            \[
                (\theta, y_{S}, z_{S}) \mapsto (y_{S} \sin(\theta), y_{S} \cos(\theta), z_{S})
            \]
            \[
                \det\left( J_{\Phi} = \begin{bmatrix} y_{S} \cos(\theta) & \sin(\theta) & 0 \\ -y_{S} \sin(\theta) & \cos(\theta) & 0 \\ 0 & 0 & 1 \end{bmatrix} \right) = y_{S}
            \]
            Di conseguenza, l'integrale diventa
            \[
                \iiint_{D} \: dx \: dy \: dz = \iint_{(y_{S}, z_{S}) \in S} \int_{0}^{2\pi} y_{S} \: d\theta \: dA = 2\pi \iint_{(y_{S}, z_{S}) \in S} y_{S} \: dA
            \]
            Analogamente:
            \begin{itemize}
                \item Se $S$ è una superficie giacente sul piano $yz$ e viene ruotata attorno all'asse $y$, allora il volume è dato da $\displaystyle 2\pi \iint_{S} z_{S} \: dA$
                \item Se $S$ è una superficie giacente sul piano $xy$ e viene ruotata attorno all'asse $x$, allora il volume è dato da $\displaystyle 2\pi \iint_{S} y_{S} \: dA$
                \item Se $S$ è una superficie giacente sul piano $xy$ e viene ruotata attorno all'asse $y$, allora il volume è dato da $\displaystyle 2\pi \iint_{S} x_{S} \: dA$
                \item Se $S$ è una superficie giacente sul piano $xz$ e viene ruotata attorno all'asse $x$, allora il volume è dato da $\displaystyle 2\pi \iint_{S} z_{S} \: dA$
                \item Se $S$ è una superficie giacente sul piano $xz$ e viene ruotata attorno all'asse $z$, allora il volume è dato da $\displaystyle 2\pi \iint_{S} x_{S} \: dA$
            \end{itemize}
