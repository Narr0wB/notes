
\section{Introduzione}
    \begin{definition}[Definizione di segnale]
        Un segnale è una qualsiasi funzione di variabili indipendenti, ovvero
        \[
            f: \Omega \subseteq \mathbb{R}^{n} \rightarrow \mathbb{R}^{m}
        \]
        \[
            \mathbf{t} \in \Omega \mapsto f(\mathbf{t})
        \]
    \end{definition}

    Da un punto di vista fisico, un segnale è \textit{una variazione di una precisa quantità fisica in funzione di tempo, spazio, o qualsiasi altra variabile indipendente}.
    Noi lavoreremo per la maggior parte con segnali tempo-varianti.

    Determinato il significato, matematico e fisico, del segnale, possiamo procedere a definire una serie di elementi utili nell'analisi di quest'ultimi.

    \begin{definition}[Definizione di supporto]
        Sia $f: \Omega \subseteq \mathbb{R} \rightarrow \mathbb{R}$ un segnale.
        Si dice supporto di $f$ il seguente insieme
        \[
            S(f) = \operatorname{cl}({\{t \in \Omega \st f(t) \neq 0\}})
        \]
    \end{definition}

    In altre parole, il supporto del segnale è la chiusura (nel senso topologico) dell'insieme degli elementi in cui il segnale non è nullo.

    \begin{definition}[Definizione di segnale localmente sommabile]
        Sia $f: \Omega \subseteq \mathbb{R} \rightarrow \mathbb{R}$ un segnale.
        Questi si dice localmente sommabile se $\forall \: a < b < \infty \in \Omega$, si ha che
        \[
            \int_{a}^{b} \abs{f(t)} \: dt < \infty
        \]
    \end{definition}

    \begin{definition}[Definizione di segnale sommabile]
        Sia $f: \Omega \subseteq \mathbb{R} \rightarrow \mathbb{R}$ un segnale.
        Questi si dice sommabile, o assolutamente integrabile, se
        \[
            \int_{\Omega} \abs{f(t)} \: dt < \infty
        \]
    \end{definition}

    \subsection{Prerequisiti}
        \begin{definition}[base di Hamel]
            Sia $V$ uno spazio vettoriale e sia $B \subseteq V$ un sottoinsieme di vettori linearmente indipendenti in $V$.
            $B$ è detto una base di Hamel qualora, per qualsiasi $\mathbf{v} \in V$ esiste una combinazione linear finita di elementi di $B$ tale che
            \[
                \mathbf{v} = \sum_{i = 1}^{k} \alpha_i \mathbf{b}_i, \; \mathbf{b}_i \in B
            \]
        \end{definition}

        \begin{definition}[base di Hilbert-Schauder]
            Sia $V$ uno spazio metrico. Un sottoinsieme $B \subseteq V$ contabile è detto base di Hilber-Schauder qualora l'insieme di tutte le combinazioni lineari finite di elementi di $B$
            sia denso in $V$.
        \end{definition}

        Uno degli spazi di interesse in cui si trovano le funzioni che ammettono serie di Fourier è $L^2([a, b]; \mathbb{C})$ dotato del seguente prodotto scalare
        \[
            f, g \in L^2([a, b]; \mathbb{C}) \implies \langle f, g \rangle = \int_{a}^{b} f(t) \overline{g(t)} dt
        \]
        Grazie a tale prodotto scalare è possibile definire una norma su $L^2([a, b]; \mathbb{C})$
        \[
            ||f||_{L^2([a, b])} = \sqrt{\langle f, f \rangle}
        \]

        Equivalentemente, posto $T = b - a$, ogni $f \in L^2([a, b]; \mathbb{C})$ può essere estesa per periodicità a tutto $\mathbb{R}$.
        Si ottiene così lo spazio $L^2(\mathbb{R}/T\mathbb{Z}; \mathbb{C})$ delle funzioni $x:\mathbb{R}\to\mathbb{C}$ $T$-periodiche e a quadrato integrabile sul periodo
        \[
            L^2(\mathbb{R}/T\mathbb{Z}; \mathbb{C}) = \left\{ x : \mathbb{R} \to \mathbb{C} \;\middle|\;
            x(t+T) = x(t) \text{ q.o.},\; \int_a^{a + T} |x(t)|^2\, dt < \infty \right\}
        \]
        dotato dello stesso prodotto scalare (calcolato su un qualsiasi intervallo di lunghezza $T$), il quale si identifica in modo naturale con $L^2([a, b]; \mathbb{C})$.
        Per tale motivo la serie di Fourier di un segnale su $[a, b]$ è a tutti gli effetti la serie di Fourier di un segnale $T$-periodico.\label{def:L2-periodic}

        \begin{definition}[Energia di un segnale]
            Sia $f \in L^2([a, b]; \mathbb{C})$ un segnale.
            L'energia di $f$ è la quantità definita come
            \[
                E(f) = ||f||^{2}_{L^2([a, b])} = \langle f, f \rangle = \int_{a}^{b} f(t) \overline{f(t)} \; dt
            \]
        \end{definition}

        \begin{definition}[Potenza di un segnale]
            Sia $f \in L^2([a, b]; \mathbb{C})$ un segnale.
            La potenza di $f$ è la quantità definita come
            \[
                P(f) = \frac{E(f)}{T}
            \]
            con $T = b - a$. 
        \end{definition}


    \subsection{Definizione}
        Data una funzione $f \in L^{2}([a, b]; \mathbb{C})$, per costruzione dello spazio $L^{2}([a, b]; \mathbb{C})$, è possibile scomporla secondo la seguente base ortogonale di Hilbert
        \[
            \{e^{in 2 \pi f_0 t}\}_{n \in \mathbb{Z}} \implies f(t) = \sum_{n \in \mathbb{Z}} a_{n} e^{in 2 \pi f_0 t}
        \]
        con $f_0 = \frac{1}{T}$, $T = b - a$ chiamata frequenza fondamentale. I coefficienti $a_n$ sono dati da
        \[
            a_{n} = \frac{1}{T} \langle f, e^{in 2 \pi f_0 t} \rangle = \frac{1}{T} \int_{a}^{b} f(t) \overline{e^{in 2 \pi f_0 t}} dt
        \]

        \begin{proposition}[Identità di Parseval]
            Sia $f: [a, b] \rightarrow \mathbb{C}$ un segnale e siano $\{a_n\}_{n \in \mathbb{Z}}$ i coefficienti della sua serie di Fourier.
            Allora l'energia di $f$ è
            \[
                E(f) = T \sum_{n \in \mathbb{Z}} |a_n|^2
            \]
        \end{proposition}

        \begin{proof}[Dimostrazione]
            Partendo dalla definizione di energia, si ha che
            \[
                E(f) = \int_{a}^{b} f(t) \overline{f(t)} dt = \int_{a}^{b} \qty(\sum_{n \in \mathbb{Z}} a_n e^{in 2\pi f_0 t}) \qty(\sum_{n \in \mathbb{Z}} \overline{a_n e^{in 2\pi f_0 t}}) dt = \sum_{i, j \in \mathbb{Z}} \int_{a}^{b} a_i \overline{a_j} e^{i (i - j) 2 \pi f_0 t} dt
            \]
            per ortonormalità della base $\{e^{in 2 \pi f_0 t}\}_{n \in \mathbb{Z}}$ si ha che per ogni $i, j \in \mathbb{Z} \st i \neq j$
            \[
                \int_{a}^{b} a_i \overline{a_j} e^{i (i - j) 2 \pi f_0 t} dt = 0
            \]
            quindi
            \[
                \sum_{i, j \in \mathbb{Z}} \int_{a}^{b} a_i \overline{a_j} e^{i (i - j) 2 \pi f_0 t} dt = \sum_{k \in \mathbb{Z}} a_k \overline{a_k} \int_{a}^{b} 1 \: dt = T \sum_{k \in \mathbb{Z}} |a_k|^2
            \]
        \end{proof}


    \subsection{Estensione a funzioni $T$-periodiche}
        \begin{theorem}[Teorema della serie di Fourier]\label{thm:fourier_series}
            Sia $f \in L^2([a, b]; \mathbb{C})$ un segnale e sia $T = b - a$ e $f_0 = \frac{1}{T}$, $f_T \in L^2(\mathbb{R}/T\mathbb{Z}; \mathbb{C})$ la sua estensione $T$-periodica in $\mathbb{R}$.
            Allora, posto
            \[
                a_n = \frac{1}{T} \int_{a}^{b} f(t) e^{-i 2 \pi n f_0 t} dt, \; n \in \mathbb{Z}
            \]
            segue che
            \[
                f_T = \sum_{n \in \mathbb{Z}} a_n e^{i 2 \pi n f_0 t} 
            \]
            quasi ovunque e la serie converge in $L^2(\mathbb{R}/T\mathbb{Z}; \mathbb{C})$.
        \end{theorem}

    \subsection{Definizione}
        \begin{lemma}
            $L^1 \cap L^2$ è denso in $L^2$.
        \end{lemma}

        \begin{proof}[Dimostrazione]
            Sia $f \in L^2$ e sia $f_{n} = f \mathbf{1}_{[-n, n]}$
            \[
                \int_{\mathbb{R}} \left| f_{n} \right| \: dt \leq \infty, \forall n \in \mathbb{N} \implies f_{n} \in L^1 \cap L^2
            \]

            Si consideri allora
            \[
                \lim_{n \rightarrow \infty} ||f - f_{n}||_{L^2} = \lim_{n \rightarrow \infty} \int_{\mathbb{R}} |f - f_{n}|^2 dt = \lim_{n \rightarrow \infty} \int_{-\infty}^{-n} |f|^2 dt + \int_{n}^{\infty} |f|^2 dt = 0
            \]
        \end{proof}

        \begin{definition}[Pre-trasformata di Fourier]
            Sia $\mathcal{F}_0: L^1 \cap L^2 \rightarrow L^{\infty}$ l'operatore definito come
            \[
                (\mathcal{F}_0 x) (f) = \int_{\mathbb{R}} x(t) e^{-i 2 \pi f t} dt
            \]
        \end{definition}

        \begin{definition}[Operatore limitato]
            Sia $T: X \rightarrow Y$ un'operatore su due spazi normati $X$ e $Y$.
            $T$ è detto limitato qualora, per ogni $x \in X$, esiste un $C \in \mathbb{R}$ tale per cui $||Tx||_{Y} \leq C||x||_{X}$.
        \end{definition}

        \begin{definition}[Operatore isometrico]
            Sia $T: X \rightarrow Y$ un'operatore su due spazi normati $X$ e $Y$.
            $T$ è detto un'isometria qualora $||Tx||_{Y} = ||x||_{X}$.
        \end{definition}

        \begin{theorem}[Teorema di Plancherel]
            Per qualsiasi $f \in L^1 \cap L^2$, si ha che $\mathcal{F}_0 f \in L^2$ e che $||\mathcal{F}_0 f||_{L^2} = ||f||_{L^2}$.
        \end{theorem}

        \begin{theorem}[Teorema della trasformazione lineare limitata (BLT)]
            Sia $X$ uno spazio normato e sia $D \subset X$ denso in $X$. 
            Sia $Y$ uno spazio di Banach, e sia $T: D \rightarrow Y$ un'operatore lineare limitato.
            Allora esiste un'unica estensione lineare limitata $\tilde{T}: X \rightarrow Y$ con $\tilde{T}|_{D} = T$. 
            Inoltre, qualora $T$ dovesse essere un'isometria, allora lo è anche $\tilde{T}$.
        \end{theorem}

        \begin{definition}[Trasformata di Fourier]
            Sia $\mathcal{F}: L^2 \rightarrow L^2$ l'estensione BLT di $\mathcal{F}_0$ tale per cui
            \[
                \mathcal{F} x = \hat{x} = X(f) = \lim_{R \rightarrow \infty} \int_{-R}^{R} x(t) e^{-i 2 \pi f t} dt
            \]
        \end{definition}

        \begin{lemma}[Antitrasformata di Fourier]
            Sia $\hat{x}$ la trasformata di Fourier di $x$, allora
            \[
                (\mathcal{F}\hat{x})(t) = x(-t) \implies \mathcal{F}^{-1} \hat{x} = (\mathcal{F} \hat{x})(-t)
            \]
        \end{lemma}

        \begin{property}[Linearità]
            $\mathcal{F}\{\alpha x + \beta y\} = \alpha \mathcal{F} x + \beta \mathcal{F} y$
        \end{property}

        \begin{property}[Traslazione]\label{prop:shift}
            $\mathcal{F}\{x(t - T)\} = (\mathcal{F}x)e^{-i 2 \pi f T}$
        \end{property}
        \begin{proof}
            \[
                \mathcal{F}\{x(t - T)\} = \int_{\mathbb{R}} x(t - T) e^{-i 2 \pi f t} dt = \int_{\mathbb{R}} x(u) e^{-i 2 \pi f (u + T)} du = (\mathcal{F}x)e^{-i 2 \pi f T} 
            \]   
        \end{proof}

        \begin{property}[Modulazione]
            $\mathcal{F}\{x(t) e^{i 2 \pi f_0 t}\} = (\mathcal{F}x)(f - f_0)$
        \end{property}
        \begin{proof}
            \[
                \mathcal{F}\{x(t) e^{i 2 \pi f_0 t}\} = \int_{\mathbb{R}} x(t) e^{i 2 \pi f_0 t} e^{-i 2 \pi f t} dt = \int_{\mathbb{R}} x(t) e^{-i 2 \pi (f - f_0) t} dt = (\mathcal{F}x)(f - f_0)
            \] 
        \end{proof}

        \begin{property}[Scalamento]
            $\mathcal{F}\{x(kt)\} = \frac{1}{|k|} (\mathcal{F}x)\qty(\frac{f}{k})$
        \end{property}
        \begin{proof}
            \[
                \mathcal{F}\{x(kt)\} = \int_{-\infty}^{\infty} x(kt) e^{-i 2 \pi f t} dt = 
                \begin{cases}
                    \int_{-\infty}^{\infty} x(u) e^{-i 2 \pi \frac{f}{k} u} \frac{du}{k}, &k > 0\\
                    \int_{\infty}^{-\infty} x(u) e^{-i 2 \pi \frac{f}{k} u} \frac{du}{k}, &k < 0
                \end{cases}
            \]
            \[
                = \int_{-\infty}^{\infty} x(u) e^{-i 2 \pi \frac{f}{k} u} \frac{du}{|k|}
                = \frac{1}{|k|} (\mathcal{F}x)\qty(\frac{f}{k})
            \]
        \end{proof}

        \begin{property}[Simmetria hermitiana]
            Sia $\hat{x} = \mathcal{F} x$.
            Se $x \in L^2$ e $x$ è reale allora $\hat{x}(-f) = \overline{\hat{x}(f)}$.
        \end{property}
        \begin{proof}
            \[
                \overline{\hat{x}(f)} = \overline{\int_{\mathbb{R}} x(t) e^{-i 2 \pi f t} dt} = \int_{\mathbb{R}} x(t) e^{i 2 \pi f t} dt = \hat{x}(-f)
            \] 
        \end{proof}

        \begin{property}[Convoluzione]\label{prop:conv}
            Siano $x_1, x_2 \in L^1$, allora $\mathcal{F}\{x_1 * x_2\} = (\mathcal{F} x_1)(\mathcal{F} x_2)$

            \begin{align*}
                \mathcal{F}\{x_1 * x_2\}(f)

            \end{align*}
        \end{proof}

    \subsection{Trasformata di segnali periodici}\label{sec:periodic_transform}
        Sia $L^2(\mathbb{R}/T\mathbb{Z})$ lo spazio dei segnali $T$-periodici a quadrato integrabile sul periodo introdotto nella sezione \ref{def:L2-periodic}.
        Si consideri la seguente immersione continua e iniettiva
        \[
            L^2(\mathbb{R}/T\mathbb{Z}) \hookrightarrow \mathcal{S}'(\mathbb{R}),
            \; \text{dove} \;
            \langle x, \varphi \rangle = \int_{\mathbb{R}} x(t)\, \varphi(t)\, dt,
            \quad \varphi \in \mathcal{S}(\mathbb{R}, \mathbb{C}).
        \]
        Di conseguenza, dato un segnale $x \in L^2(\mathbb{R}/T\mathbb{Z})$, per il teorema della serie di Fourier (\Cref{thm:fourier_series}), vale
        \[
            x = \sum_{n \in \mathbb{Z}} a_{n} e^{i 2 n \pi f_{0} t} \; \text{in} \; L^2(\mathbb{R}/T\mathbb{Z})
        \]
        Dato che l'immersione da $L^2(\mathbb{R}/T\mathbb{Z})$ a $\mathcal{S}'$ è continua, si ha che
        \[
            x = \sum_{n \in \mathbb{Z}} a_{n} e^{i 2 n \pi f_{0} t} \; \text{in} \; \mathcal{S}'
        \]
        Segue allora che l'espressione $\mathcal{F}x$ è ben definita e dato che l'operatore di Fourier è continuo in $\mathcal{S}'$, che
        \[
            \mathcal{F}x = \mathcal{F} \qty{ \sum_{n \in \mathbb{Z}} a_{n} e^{i 2 n \pi f_{0} t} } 
            = \sum_{n \in \mathbb{Z}} a_n \mathcal{F} \{ e^{i 2 n \pi f_0 t} \}
            = \sum_{n \in \mathbb{Z}} a_n \delta(f - n f_0)
        \]


    \subsection{Operatore di autocorrelazione}
        \begin{definition}[Autocorrelazione]
            L'operatore di autocorrelazione $\mathcal{R}: L^2 \rightarrow L^\infty$ è definito come
            \[
                (\mathcal{R} x)(t) := \int_{\mathbb{R}} x(\tau) x^*(\tau - t) d\tau = x * Sx^*
            \]
            dove $S: L^2 \rightarrow L^2$ è l'operatore specchio ($Sx = x(-t)$).
        \end{definition}

        \begin{proposition}
            $\mathcal{R} x = \mathcal{F}^{-1}\{ |X(f)|^2 \}$ 
        \end{proposition}
        \begin{proof}
            Partendo dalla definizione di $\mathcal{R} x = x * (Sx^*)$, grazie al teorema della convoluzione (Proprietà 6 della trasformata di Fourier)
            \[
                \mathcal{R} x = x * (Sx^*) = \mathcal{F}^{-1} \{ (\mathcal{F} x)((\mathcal{F} \circ S) x^*) \}
            \]
            Si analizzi il termine $(\mathcal{F} \circ S) x^*$
            \[
                (\mathcal{F} \circ S)x^*
                = \int_{\mathbb{R}} x^*(-t)\, e^{-i2\pi f t}\, dt
                = \left( \int_{\mathbb{R}} x(-t)\, e^{i2\pi f t}\, dt \right)^*
                \overset{s=-t}{=} \left( \int_{\mathbb{R}} x(s)\, e^{-i2\pi f s}\, ds \right)^*
                = X^*(f)
            \]
            Arrivando quindi alla conclusione
            \[
                \mathcal{R} x = \mathcal{F}^{-1} \{ (\mathcal{F} x)((\mathcal{F} \circ S) x^*) \} = \mathcal{F}^{-1} \{ X(f) X^*(f) \} = \mathcal{F}^{-1} \{ |X(f)|^2 \}
            \]
        \end{proof}


        Alcune delle proprietà dell'operatore di autocorrelazione sono
        
        \begin{property}[Energia]
            L'energia di un segnale $x \in L^2$ vale $||x||^2_2 = (\mathcal{R}x)(0)$
        \end{property}
        \begin{proof}
            Valutando la definizione in $t = 0$
            \[
                (\mathcal{R}x)(0) = \int_{\mathbb{R}} x(\tau) x^*(\tau) d\tau = \int_{\mathbb{R}} |x(\tau)|^2 d\tau = ||x||^2_2
            \]
        \end{proof}

        \begin{property}[Simmetria]
            Sia $x \in L^2$, allora $S(\mathcal{R}x) = (\mathcal{R}x)^*$, ovvero $\mathcal{R}x$ è una funzione hermitiana.
            In particolare, se $x$ è reale allora $S(\mathcal{R}x) = \mathcal{R}x$, ovvero $\mathcal{R}x$ è una funzione pari.
        \end{property}
        \begin{proof}
            \begin{align*}
                (S\mathcal{R}x)(t) &= (\mathcal{R}x)(-t) = \int_{\mathbb{R}} x(\tau) x^*(\tau + t) d\tau
                \overset{u = \tau + t}{=} \int_{\mathbb{R}} x(u - t) x^*(u) du \\
                &= \left( \int_{\mathbb{R}} x(u) x^*(u - t) du \right)^*
                = (\mathcal{R}x)^*(t)
            \end{align*}
            Se $x$ è reale, anche $\mathcal{R}x$ è reale, dunque $(\mathcal{R}x)^* = \mathcal{R}x$ e quindi $S(\mathcal{R}x) = \mathcal{R}x$.
        \end{proof}

        \begin{property}[Massimo nell'origine]
            Sia $x \in L^2$, allora $|(\mathcal{R}x)(t)| \leq (\mathcal{R}x)(0)$ per ogni $t \in \mathbb{R}$.
            In particolare, se $\mathcal{R}x$ è reale, $(\mathcal{R}x)(t) \leq (\mathcal{R}x)(0)$.
        \end{property}
        \begin{proof}
            Per la disuguaglianza di Cauchy-Schwarz in $L^2$
            \[
                |(\mathcal{R}x)(t)| = \left| \int_{\mathbb{R}} x(\tau) x^*(\tau - t) d\tau \right|
                \leq \qty( \int_{\mathbb{R}} |x(\tau)|^2 d\tau )^{\frac{1}{2}} \qty( \int_{\mathbb{R}} |x(\tau - t)|^2 d\tau )^{\frac{1}{2}}
                = ||x||_2 \, ||x||_2 = (\mathcal{R}x)(0)
            \]
            Se $\mathcal{R}x$ è reale, segue $(\mathcal{R}x)(t) \leq |(\mathcal{R}x)(t)| \leq (\mathcal{R}x)(0)$.
        \end{proof}
    

    \subsection{Spettro di potenza}
        \begin{definition}[Densità spettrale di potenza]
            Sia $x \in \mathcal{S}'$ un segnale e si denoti con $x_{\tau}(t)$ la restrizione $x|_{[-\tau, \tau]}(t)$.
            Sia allora $X_{\tau}(f) = X * 2\tau \operatorname{sinc}(2\tau f)$ la trasformata di $x_{\tau}$.
            Allora la funzione di densità spettrale di potenza del segnale $x$ è definita come
            \[
                P_{x}(f) := \lim_{\tau \rightarrow \infty} \frac{|X_{\tau}(f)|^2}{2 \tau}
            \]
        \end{definition}

        Alcune delle proprietà della densità spettrale di potenza sono

        \begin{property}[Non negatività]
            $P_x(f) \geq 0$ per ogni $f \in \mathbb{R}$.
        \end{property}
        \begin{proof}
            Per ogni $\tau > 0$ si ha che $|X_{\tau}(f)|^2 \geq 0$ e $2\tau > 0$, quindi
            \[
                \frac{|X_{\tau}(f)|^2}{2 \tau} \geq 0, \; \forall \tau > 0
            \]
            Essendo il limite di una funzione non negativa, anche $P_x(f) = \lim_{\tau \rightarrow \infty} \frac{|X_{\tau}(f)|^2}{2 \tau} \geq 0$.
        \end{proof}

        \begin{property}[Autocorrelazione di un segnale a potenza finita]
            Sia $R_x$ la funzione di autocorrelazione (in media temporale) del segnale $x$
            \[
                R_x(t) := \lim_{\tau \rightarrow \infty} \frac{1}{2\tau} \int_{-\tau}^{\tau} x(s) x^*(s - t) \, ds
            \]
            Allora la densità spettrale di potenza è la trasformata di Fourier dell'autocorrelazione
            \[
                P_x = \mathcal{F} R_x, \qquad R_x = \mathcal{F}^{-1} P_x
            \]
        \end{property}

\section{Sistemi}
    Un sistema è una mappa $\mathcal{H}: \mathcal{S}' \rightarrow \mathcal{S}'$ che associa ad un segnale in ingresso un altro segnale in uscita.

    \begin{figure}[ht]
        \centering
        \begin{tikzpicture}[>={Stealth[length=2.5mm]}]
            \node[draw, thick, minimum width=2cm, minimum height=1.2cm] (H) {$\mathcal{H}$};
            \draw[->, thick] ([xshift=-2cm]H.west) -- node[above] {$x(t)$} (H.west);
            \draw[->, thick] (H.east) -- node[above] {$y(t)$} ([xshift=2cm]H.east);
        \end{tikzpicture}
        \caption{Rappresentazione a scatola nera di un sistema $\mathcal{H}$, con $y = \mathcal{H} x$.}
        \label{fig:sistema}
    \end{figure}

    I sistemi sono caratterizzati dalle rispettive proprietà:
    \begin{enumerate}
        \item Lineare: $\mathcal{H}\{\alpha x_1 + \beta x_2\} = \alpha \mathcal{H}\{x_1\} + \beta \mathcal{H}\{x_2\}$
        \item Tempo-invariante: $\mathcal{H}\{x(t - \tau)\} = \mathcal{H}\{x\}(t - \tau)$
    \end{enumerate}

    \begin{definition}[Memoria di un sistema]
        Un sistema $\mathcal{H}$ si dice dotato di memoria qualora l'uscita del sistema $y(t) := \mathcal{H} x$ ad un dato istante $t$ dipenda dall'ingresso $x(t)$ valutato ad un istante $t_0 < t$.
    \end{definition}

    Un sistema avente proprietà $(1)$ e $(2)$ è detto \textbf{sistema LTI} (dall'inglese \textit{linear time-invariant}), e si dimostra che è possibile ricondurre ogni operatore LTI $\mathcal{H}$ a
    \[
        \mathcal{H} x = x * h = \int_{\mathbb{R}} x(t - \tau) h(\tau) d\tau \implies \mathcal{F}\{ \mathcal{H} x \} = X(f) H(f)
    \]
    dove $(\cdot) * (\cdot): \mathcal{S}' \times \mathcal{S}' \rightarrow \mathcal{S}'$ è l'operatore di convoluzione e $h(t)$ è detta \textit{risposta all'impulso} del sistema $\mathcal{H}$. 
    D'ora in poi il focus dell'attuale sezione saranno i sistemi LTI, a meno che non venga segnalato diversamente.

    \begin{definition}[Stabilità \textit{Bounded Input Bounded Output}]
        Un sistema $\mathcal{H}$ è detto BIBO-stabile se
        \[
            \max_{t \in \mathbb{R}} |x(t)| < \infty \implies \max_{t \in \mathbb{R}} |(\mathcal{H} x)(t)| < \infty
        \]
    \end{definition}

    \begin{proposition}
        Un sistema $\mathcal{H}$ è BIBO-stabile se e solo se $h(t) \in L^1$.
    \end{proposition}
    \begin{proof}
        ($\Leftarrow$) Si consideri la disuguaglianza di Young con $p = 1$ e $q = 2$
        \[
            ||x * h||_{2} \leq ||h||_{1} ||x||_{2}
        \]

    \end{proof}

    \begin{definition}[Banda di un sistema]
        Si definisce la banda di un sistema $\mathcal{H}$ come segue:
        \begin{itemize}
            \item Se $H(f)$ ha supporto limitato, allora la banda $B$ diventa $\frac{\mu(\operatorname{supp}(H(f)))}{2}$.
            \item Se $H(f)$ ha supporto illimitato, si definisce la banda a $-3dB$ ($B_{-3dB}$), una volta definito l'insieme $D \subset \operatorname{supp}(H(f))$ tale per cui $|H(D)| \geq \frac{1}{\sqrt{2}} |H(0)|$, come
            \[
                B_{-3dB} = \mu(D)
            \]
        \end{itemize}
    \end{definition}

    \subsection{Causalità di un sistema e real-time signal processing}
        \begin{definition}[Sistema causale]
            Un sistema LTI è detto causale quando la sua risposta all'impulso $h(t) = 0$ per ogni $t < 0$.
        \end{definition}

        La definizione di sistema causale è necessaria nel dominio del \textit{real-time signal processing}, ovvero lo studio dei segnali in tempo reale.
        Si ipotizzi di avere a disposizione un segnale $x|_{(-\infty, t_0]}$ fino ad un dato istante $t_0$, ovvero $x|_{(-\infty, t_0]}(t) = 0, \forall t > t_0$. L'uscita del segnale in $t_0$ è data da
        \[
            \mathcal{H}\{x|_{(-\infty, t_0]}\}(t_0) = x|_{(-\infty, t_0]} * h = \int_{-\infty}^{0} x|_{(-\infty, t_0]}(t_0 - \tau) h(\tau) d\tau + \int_{0}^{\infty} x|_{(-\infty, t_0]}(t_0 - \tau) h(\tau) d\tau
        \]
        si può osservare come il primo integrale dipenda da valori futuri di $x|_{(-\infty, t_0]}$, ovvero oltre $t_0$, in quanto $\tau < 0 \implies - \tau > 0 \implies t_0 - \tau > t_0$ che però non esistono ancora.
        Se $h$ non fosse causale, si avrebbe che $\mathcal{H}\{x|_{(-\infty, t_0]}\}(t_0) \neq \mathcal{H}\{x|_{\mathbb{R}}\}(t_0)$.

    \subsection{Distorsione}
        Un sistema \textbf{non} affetto da distorsione è un sistema tale per cui l'uscita, dato un segnale $x(t)$, è una versione scalata e ritardata dell'ingresso
        \[
            (\mathcal{H} x)(t) = K x(t - t_0), \quad K \neq 0, t_0 \geq 0
        \]
        Il che implica
        \[
            \mathcal{F}\{ K x(t - t_0) \} = K X(f) e^{-i 2 \pi f t_0} \implies H(f) = K e^{-i 2 \pi f t_0}
        \]
        In altre parole, il sistema non-distorcente per definizione ha modulo costante e fase lineare.

        \begin{definition}[Distorsione di modulo]
            Un sistema $\mathcal{H}$ è affetto da distorsione di modulo qualora
            \[
                \frac{d }{df} |H(f)| \neq 0 
            \]
        \end{definition}

        \begin{definition}[Distorsione di fase]
            Un sistema $\mathcal{H}$ è affetto da distorsione di fase qualora
            \[
                \frac{d}{df} \arg(H(f)) \neq K, \quad K \in \mathbb{R}
            \]
        \end{definition}

\section{Segnali discreti}
    \begin{definition}[Segnale discreto]
        Per segnale discreto si intente una qualsiasi successione del tipo 
        \[
            \{x_n\}_{n \in N \subseteq \mathbb{N}}
        \]
        La notazione per un segnale discreto è
        \[
            x[n], \; n \in N \subseteq \mathbb{N}
        \]
    \end{definition}

    \subsection{Campionamento periodico di segnali continui}
        Sia $x \in C_{0}(\mathbb{R})$ un segnale continuo, si definisce la sua versione campionata a frequenza $f_s = \frac{1}{T_s}$ in $\mathcal{S}'$ il segnale determinato da
        \[
            \tilde x = T_{s} \sum_{n \in \mathbb{Z}} x(n T_s) \delta(t - n T_s)
        \]
        Definito il pettine di Dirac di periodo $T_s$
        \[
            \operatorname{III}_{T_s}(t) := \sum_{n \in \mathbb{Z}} \delta(t - n T_s)
        \]
        e osservando che $x(t) \delta(t - n T_s) = x(n T_s) \delta(t - n T_s)$, il segnale campionato si può riscrivere come il prodotto tra $x$ e il pettine
        \[
            \tilde x = T_s \sum_{n \in \mathbb{Z}} x(t) \delta(t - n T_s) = T_s \, x(t) \operatorname{III}_{T_s}(t)
        \]
        Si osservi allora la trasformata di $\tilde x$: per dualità del teorema della convoluzione (P\ref{prop:conv}), la trasformata di un prodotto è la convoluzione delle trasformate, quindi
        \[
            \tilde X(f) = T_s \, \mathcal{F}\{ x \operatorname{III}_{T_s} \} = T_s \, (X * \mathcal{F}\{ \operatorname{III}_{T_s} \})
        \]
        Rimane da calcolare la trasformata del pettine.
        Essendo $\operatorname{III}_{T_s}$ un segnale $T_s$-periodico, se ne calcolino (formalmente) i coefficienti di Fourier su un periodo centrato nell'origine
        \[
            a_k = \frac{1}{T_s} \int_{-\frac{T_s}{2}}^{\frac{T_s}{2}} \delta(t) e^{-i 2 \pi k f_s t} dt = \frac{1}{T_s} = f_s, \; k \in \mathbb{Z}
        \]
        da cui, analogamente a quanto visto in \cref{sec:periodic_transform} (con la serie che converge in $\mathcal{S}'$),
        \[
            \operatorname{III}_{T_s}(t) = f_s \sum_{k \in \mathbb{Z}} e^{i 2 \pi k f_s t}
            \implies
            \mathcal{F}\{ \operatorname{III}_{T_s} \}(f) = f_s \sum_{k \in \mathbb{Z}} \delta(f - k f_s) = f_s \operatorname{III}_{f_s}(f)
        \]
        ovvero la trasformata di un pettine di periodo $T_s$ è un pettine di periodo $f_s$, scalato di $f_s$.
        Riassemblando, e ricordando che $T_s f_s = 1$ e che $X * \delta(f - k f_s) = X(f - k f_s)$, si ottiene
        \[
            \tilde X(f) = T_s \, X(f) * f_s \sum_{k \in \mathbb{Z}} \delta(f - k f_s)
            = \sum_{k \in \mathbb{Z}} X(f) * \delta(f - k f_s)
            = \sum_{k \in \mathbb{Z}} X(f - k f_s)
        \]
        ovvero lo spettro del segnale campionato è la periodicizzazione di $X$ con periodo $f_s$.


        \begin{theorem}[Teorema del campionamento]
            Dato un segnale $x \in \mathcal{S}'$ reale a banda limitata $B$, è possibile ricostruire il segnale $x$ a partire da un campionamento
            \[
                \{x(n T_s)\}_{n \in \mathbb{Z}}
            \]
            a patto che $f_s = \frac{1}{T_s} > 2B$.
        \end{theorem}

        \subsubsection{Aliasing}
            Il fenomeno dell'aliasing consiste nell'ottenere, da un segnale campionato, un segnale che è diverso dal segnale originale, e si ottiene quando si campiona un segnale la cui banda $B$ si estende oltre il limite di Nyquist $\frac{f_s}{2}$.

            Una principale soluzione all'aliasing è introdurre, prima di campionare, un filtro detto ``anti-aliasing'', in grado di eliminare (nella realtà attenuare) tutte le frequenze oltre il limite di Nyquist del campionatore.


    \subsection{Discrete Fourier Transform (DFT)}
        Il \textit{Discrete Fourier Transform}, in italiano Trasformata di Fourier Discreta, è una tecnica che permette di calcolare uno spettro discreto a partire da un segnale discreto a campioni finiti.

        Si consideri di avere un segnale campionato finito $x[n]$ con $0 \leq n < N$, campionato alla frequenza $f_s = \frac{1}{T_s}$ e
        si definisca la finestra di osservabilità come $N T_s$ e lo step in frequenza come $\Delta f = \frac{f_s}{N}$.
        Il DFT mappa questo vettore di campioni $x[n]$ ad uno spettro $X[k]$ con $0 \leq k < N$ definito nel seguente modo
        \[
            X[k] = \sum_{n=0}^{N-1} x[n] e^{-i 2 \pi \frac{kn}{N}}
        \]
        La mappa campione $\rightarrow$ spettro introdotta dal DFT è iniettiva, quindi è possibile ricostruire il vettore di campioni a partire dal vettore di spettro
        \[
            x[n] = \frac{1}{N} \sum_{k=0}^{N-1} X[k] e^{i 2 \pi \frac{kn}{N}}
        \]

    \subsection{Sistemi a tempo discreto}
        Un sistema a tempo discreto è un oggetto matematico in grado di mappare un vettore discreto $x[n]$ in ingresso ad un vettore discreto $y[n]$ in uscita.
        \begin{figure}[ht]
            \centering 
            \begin{tikzpicture}
                \node[draw, thick, minimum width=2cm, minimum height=1.2cm] (H) {$\mathcal{H}$};
                \draw[->, thick] ([xshift=-2cm]H.west) -- node[above] {$x[n]$} (H.west);
                \draw[->, thick] (H.east) -- node[above] {$y[n]$} ([xshift=2cm]H.east);
            \end{tikzpicture}
        \end{figure}
        
        un sistema a tempo discreto può essere un sitema linear tempo-invariante (LTI), con 
        \[
            \mathcal{H}\{x[n]\} = x[n] * h[n] = \sum_{k \in \mathbb{Z}} h[n - k] x[k]
        \]
        dove $h[n]$ è una risposta all'impulso del sistema a tempo discreto.


    \subsection{Trasformata Z}
        \begin{definition}
            Dato un segnale discreto $x[n]$ con $n \in \mathbb{Z}$, si definisce la trasformata Z di $x[n]$ come
            \[
                X(z) = \mathcal{Z}\{x\} = \sum_{n \in \mathbb{Z}} x[n] z^{-n}, \qquad z \in \mathcal{R}_{x} \subseteq \mathbb{C}
            \]
            con 
            \[
                \mathcal{R}_x = \qty{ z \in \mathbb{C} \st \sum_{n \in \mathbb{Z}} |x[n] z^{-n}| < \infty}
            \]
        \end{definition}

        La trasformata Z rapprensenta il corrispettivo discreto della trasformata di Laplace nel continuo.
        L'insieme $\mathcal{R}_{x}$ su cui la trasformata Z converge per un dato segnale $x$ è detto regione di convergenza.
        Inoltre è possibile dimostrare che $(1)$ $\mathcal{R}_{x}$ è un anello e che $(2)$, per segnali causali ($x[n] = 0, \forall n < 0$), che il raggio della circonferenza esterna è $\infty$.

        Alcune delle proprietà della trasformata Z sono:

        \begin{property}[Linearità]
            $\mathcal{Z}\{\alpha x[n] + \beta y[n]\} = \alpha X(z) + \beta Y(z)$, con regione di convergenza contenente $\mathcal{R}_{x} \cap \mathcal{R}_{y}$.
        \end{property}

        \begin{property}[Traslazione]
            $\mathcal{Z}\{x[n - n_0]\} = z^{-n_0} X(z)$, con regione di convergenza $\mathcal{R}_{x}$, a meno dei punti $z = 0$ e $z = \infty$.
        \end{property}

        \begin{property}[Modulazione]
            $\mathcal{Z}\{a^n x[n]\} = X\qty(\frac{z}{a})$, con $a \in \mathbb{C} \setminus \{0\}$ e regione di convergenza $|a| \mathcal{R}_{x}$.
        \end{property}

        \begin{property}[Inversione temporale]
            $\mathcal{Z}\{x[-n]\} = X\qty(\frac{1}{z})$, con regione di convergenza $\qty{ z \in \mathbb{C} \st \frac{1}{z} \in \mathcal{R}_{x} }$.
        \end{property}

        \begin{property}[Coniugazione]
            $\mathcal{Z}\{x^*[n]\} = X^*(z^*)$, con regione di convergenza $\mathcal{R}_{x}$.
        \end{property}

        \begin{property}[Derivata di trasformata]
            $\mathcal{Z}\{n x[n]\} = -z \frac{d}{dz} X(z)$, con regione di convergenza $\mathcal{R}_{x}$.
        \end{property}

        \begin{property}[Convoluzione]
            $\mathcal{Z}\{x[n] * y[n]\} = X(z) Y(z)$, con regione di convergenza contenente $\mathcal{R}_{x} \cap \mathcal{R}_{y}$.
        \end{property}

        \begin{property}[Relazione con la trasformata di Fourier]
            Se la circonferenza unitaria $|z| = 1$ è contenuta in $\mathcal{R}_{x}$, allora la trasformata di Fourier a tempo discreto di $x[n]$ si ottiene valutando $X(z)$ per $z = e^{i 2 \pi \nu}$, con $\nu = f T_s$ frequenza normalizzata
            \[
                X(e^{i 2 \pi \nu}) = \sum_{n \in \mathbb{Z}} x[n] e^{-i 2 \pi \nu n}
            \]
        \end{property}

        \begin{property}[Teorema del valore iniziale]
            Se $x[n]$ è causale, allora $x[0] = \lim_{z \to \infty} X(z)$.
        \end{property}

        \begin{property}[Teorema del valore finale]
            Se $x[n]$ è causale e i poli di $(z - 1) X(z)$ sono interni alla circonferenza unitaria, allora $\lim_{n \to \infty} x[n] = \lim_{z \to 1} (z - 1) X(z)$.
        \end{property}

    \subsection{Discrete Time Fourier Transform (DTFT)}
        \begin{definition}
            Dato un segnale discreto $x[n]$ con $n \in \mathbb{Z}$ e $x \in \ell^1$, si definisce la trasformata di Fourier a tempo discreto di $x[n]$ come
            \[
                X(e^{i 2 \pi \nu}) = \sum_{n \in \mathbb{Z}} x[n] e^{-i 2 \pi \nu n}, \qquad \nu \in \mathbb{R}
            \]
            dove $\nu = f T_s = \frac{f}{f_s}$ è detta frequenza normalizzata.
        \end{definition}

        La DTFT coincide con la trasformata Z valutata sulla circonferenza unitaria $z = e^{i 2 \pi \nu}$, ed è dunque ben definita qualora $|z| = 1$ sia contenuta in $\mathcal{R}_{x}$.
        A differenza della trasformata Z, la definizione può essere estesa a segnali $x \in \ell^2$, nel qual caso la serie converge in $L^2$.
        È possibile ricostruire il segnale a partire dalla sua DTFT mediante la trasformata inversa
        \[
            x[n] = \int_{-\frac{1}{2}}^{\frac{1}{2}} X(e^{i 2 \pi \nu}) e^{i 2 \pi \nu n} d\nu
        \]

        Alcune delle proprietà della DTFT sono:

        \begin{property}[Periodicità]
            $X(e^{i 2 \pi (\nu + 1)}) = X(e^{i 2 \pi \nu})$, ovvero la DTFT è periodica di periodo $1$ in $\nu$ (di periodo $f_s$ in $f$).
        \end{property}

        \begin{property}[Linearità]
            $\mathcal{F}\{\alpha x[n] + \beta y[n]\} = \alpha X(e^{i 2 \pi \nu}) + \beta Y(e^{i 2 \pi \nu})$
        \end{property}

        \begin{property}[Traslazione]
            $\mathcal{F}\{x[n - n_0]\} = e^{-i 2 \pi \nu n_0} X(e^{i 2 \pi \nu})$
        \end{property}

        \begin{property}[Modulazione]
            $\mathcal{F}\{e^{i 2 \pi \nu_0 n} x[n]\} = X(e^{i 2 \pi (\nu - \nu_0)})$
        \end{property}

        \begin{property}[Inversione temporale]
            $\mathcal{F}\{x[-n]\} = X(e^{-i 2 \pi \nu})$
        \end{property}

        \begin{property}[Simmetria hermitiana]
            Se $x[n]$ è reale allora $X(e^{-i 2 \pi \nu}) = X^*(e^{i 2 \pi \nu})$.
        \end{property}

        \begin{property}[Convoluzione]
            $\mathcal{F}\{x[n] * y[n]\} = X(e^{i 2 \pi \nu}) Y(e^{i 2 \pi \nu})$
        \end{property}

        \begin{property}[Trasformata di prodotto]
            Il prodotto nel tempo corrisponde alla convoluzione periodica in frequenza
            \[
                \mathcal{F}\{x[n] y[n]\} = \int_{-\frac{1}{2}}^{\frac{1}{2}} X(e^{i 2 \pi \theta}) Y(e^{i 2 \pi (\nu - \theta)}) d\theta
            \]
        \end{property}

        \begin{property}[Derivata di trasformata]
            $\mathcal{F}\{n x[n]\} = \frac{i}{2 \pi} \frac{d}{d\nu} X(e^{i 2 \pi \nu})$
        \end{property}

        \begin{property}[Parseval]
            $\displaystyle \sum_{n \in \mathbb{Z}} |x[n]|^2 = \int_{-\frac{1}{2}}^{\frac{1}{2}} |X(e^{i 2 \pi \nu})|^2 d\nu$
        \end{property}

        \begin{property}[Relazione con il campionamento]
            Se $x[n] = x_c(n T_s)$ è ottenuto campionando un segnale continuo $x_c$ con trasformata $X_c(f)$, allora
            \[
                X(e^{i 2 \pi \nu}) = f_s \sum_{k \in \mathbb{Z}} X_c\qty((\nu - k) f_s)
            \]
        \end{property}

        \begin{property}[Relazione con la DFT]
            Se $x[n]$ è non nullo solo per $0 \leq n < N$, allora la DFT è il campionamento della DTFT in $N$ punti equispaziati nel periodo
            \[
                X[k] = X(e^{i 2 \pi \frac{k}{N}}), \qquad 0 \leq k < N
            \]
        \end{property}

    \subsection{Definizione}
    \begin{definition}[Segnale stocastico]
        Un segnale stocastico, anche detto processo stocastico, è una funzione misurabile del tipo $X: \mathbb{R} \times \Omega \rightarrow \mathbb{R}$, tale per cui, dato uno spazio di probabilità $(\Omega, \mathcal{F}, P)$, si ha che
        \[
            \{\omega \in \Omega \st X(t, \omega) \leq a \} \in \mathcal{F}, \qquad t, a \in \mathbb{R}
        \]
        
        Fissato un istante $t$, la mappa $X(t, \omega): \Omega \rightarrow \mathbb{R}$ è una variabile casuale.
        Viceversa, fissato un $\omega$, la mappa $X(t, \omega): \mathbb{R} \rightarrow \mathbb{R}$ è detta realizzazione del segnale.
    \end{definition}

    Dato un processo stocastico è possibile costruire i parametri principali di una variabile casuale come mappe
    \[
        \begin{aligned}
            \mu_{X}(t) &: t \mapsto \mu_{X(t)}, & \mu_{X}(t) &= \mathbb{E}[X(t)] = \int_{\mathbb{R}} x f_{X(t)}(x) dx \\
            \sigma_{X}^2(t) &: t \mapsto \sigma_{X(t)}^2, & \sigma_{X}^2(t) &= \mathbb{E}[(X(t) - \mu_{X(t)})^2] = \mathbb{E}[X^2(t)] - \mu_{X(t)}^2 = \int_{\mathbb{R}} x^2 f_{X(t)}(x) dx - \mu_{X(t)}^2 \\
            \sigma_{X}(t) &: t \mapsto \sigma_{X(t)}, & \sigma_{X}(t) &= \sqrt{\sigma_{X}^2(t)}
        \end{aligned}
    \]
    le quali, a partire da un instante $t$, portano ai rispettivi parametri (media, varianza e deviazione standard) della variabile casuale $X(t): \Omega \rightarrow \mathbb{R}$.

    \subsection{Autocorrelazione e stazionarietà di un processo stocastico}
    \begin{definition}[Autocorrelazione di variabili casuali]
        Sia $X: \mathbb{R} \times \Omega \rightarrow \mathbb{C}$ un processo stocastico.
        Si definisce l'autocorrelazione fra due istanti $t_1, t_2$ la seguente espressione
        \[
            R_{X}(t_1, t_2) = \mathbb{E}[X(t_1)X^*(t_2)]
        \]
    \end{definition}

    \begin{definition}[Processo stocastico stazionario in senso lato]
        Un processo stocastico $X: \mathbb{R} \times \Omega \rightarrow \mathbb{C}$ è detto stazionario in senso lato (in inglese, wide sense stationary) qualora valessero le seguenti condizioni:
        \begin{enumerate}
            \item $\mathbb{E}[|X(t)|^2] < \infty$ per qualsiasi $t \in \mathbb{R}$.
            \item $\mu_{X}(t) = \mu$ per qualsiasi $t \in \mathbb{R}$, con $\mu \in \mathbb{C}$ costante.
            \item $R_{X}(t_1, t_2) = R_{X}(\tau) = \mathbb{E}[X(t)X^*(t + \tau)]$ per qualsiasi $t \in \mathbb{R}$, con $\tau = t_2 - t_1$.
        \end{enumerate}
    \end{definition}

    Alcune delle proprietà di un processo stazionario in senso lato sono:

    \begin{property}[Potenza]
        $R_{X}(0) = \mathbb{E}[|X(t)|^2]$ è reale e non negativo, e rappresenta la potenza media del processo.
    \end{property}

    \begin{property}[Simmetria]
        $R_{X}(-\tau) = R_{X}^*(\tau)$ per qualsiasi $\tau \in \mathbb{R}$, ovvero $R_{X}$ è una funzione hermitiana.
        In particolare, se il processo è reale, ovvero $X: \mathbb{R} \times \Omega \rightarrow \mathbb{R}$, allora $R_{X}$ è reale e pari, ovvero $R_{X}(-\tau) = R_{X}(\tau)$.
    \end{property}

    \begin{property}[Massimo nell'origine]
        $|R_{X}(\tau)| \leq R_{X}(0)$ per qualsiasi $\tau \in \mathbb{R}$.
    \end{property}

    \begin{property}[Varianza costante]
        $\sigma_{X}^2(t) = \mathbb{E}[X^2(t)] - \mu^2 = R_{X}(0) - \mu^2$.
    \end{property}

    \begin{definition}[Processo stocastico ciclostazionario in senso lato]
        Un processo stocastico è detto ciclostazionario in senso lato (in inglese cyclostationary) con periodo $T$ qualora valessero se seguenti condizioni:
        \begin{enumerate}
            \item $\mu_{X}(t) = \mu_{X}(t + T)$
            \item $R_{X}(t_1, t_2) = R_{X}(t_1 + T, t_2 + T)$
        \end{enumerate}
    \end{definition}

    \begin{definition}[Processo stocastico non stazionario]
        Un processo stocastico è detto non stazionario qualora le sue proprietà $\mu_{X}(t), \sigma_{X}^2(t)$ varino nel tempo.
    \end{definition}

    \subsection{Analisi in frequenza di processi stocastici WSS}
        \begin{definition}[Spettro di potenza di un processo WSS]
            Sia $X: \mathbb{R} \times \Omega \rightarrow \mathbb{C}$ un processo stocastico WSS.
            Si definisce allora il suo spettro di potenza come
            \[
                P_{X}(f) = \mathcal{F}\{R_{X}(\tau)\}
            \]
        \end{definition}

        Alcune delle proprietà dello spettro di potenza di un processo WSS sono:

        \begin{property}[Spettro reale]
            $P_{X}(f) \in \mathbb{R}$ per qualsiasi $f \in \mathbb{R}$, in quanto trasformata di una funzione hermitiana.
        \end{property}

        \begin{property}[Non negatività]
            $P_{X}(f) \geq 0$ per qualsiasi $f \in \mathbb{R}$.
        \end{property}

        \begin{property}[Simmetria]
            Se il processo è reale, ovvero $X: \mathbb{R} \times \Omega \rightarrow \mathbb{R}$, allora $P_{X}$ è una funzione pari, ovvero $P_{X}(-f) = P_{X}(f)$.
        \end{property}

        \begin{property}[Potenza]
            La potenza media del processo è pari all'integrale dello spettro di potenza
            \[
                R_{X}(0) = \mathbb{E}[|X(t)|^2] = \int_{\mathbb{R}} P_{X}(f) df
            \]
        \end{property}

    \subsection{Rumore}
        E' possibile definire il rumore in un segnale come un processo stocastico il quale viene poi sommato ad un segnale deterministico, nella forma
        \[
            X(t) = s(t) + N(t)
        \]
        Dove $X(t)$ è il segnale misurato, $s(t)$ il segnale deterministico sottostante e $N(t)$ un processo stocastico chiamato \textit{rumore}.

        Vi sono diversi tipi di rumore, fra cui:
        \subsubsection{Rumore gaussiano bianco}
            Il rumore gaussiano bianco $N(t)$ è un processo stocastico WSS definito dai seguenti parametri:
            \begin{enumerate}
                \item Media: $\mu_{N} = 0$
                \item Autocorrelazione: $R_{N}(\tau) = \frac{N_0}{2} \delta(\tau)$
            \end{enumerate}
